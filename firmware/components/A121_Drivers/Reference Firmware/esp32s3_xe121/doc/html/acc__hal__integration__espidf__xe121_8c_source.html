59 static void acc_hal_integration_sensor_transfer(acc_sensor_id_t sensor_id, uint8_t *buffer, size_t buffer_size)
60 {
61  esp_err_t ret;
62  spi_transaction_t t;
63 
64  memset(&t, 0, sizeof(t));
65  t.length = buffer_size * 8;
66  t.tx_buffer = buffer;
67  t.rx_buffer = buffer;
68  t.user = (void *)1;
69  ret = spi_device_polling_transmit(spi, &t);
70  ESP_ERROR_CHECK(ret);
71 }

73 static void isr_handler(void *args)
74 {
75  BaseType_t xHigherPriorityTaskWoken = pdFALSE;
76 
77  xSemaphoreGiveFromISR(isr_sem, &xHigherPriorityTaskWoken);
78  portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
79 }

81 bool acc_hal_integration_wait_for_sensor_interrupt(acc_sensor_id_t sensor_id, uint32_t timeout_ms)
82 
83 {
84  (void)sensor_id; // Ignore parameter sensor_id
85 
86  int interupt_level = gpio_get_level(GPIO_INTERRUPT);
87 
88  if (interupt_level == 1)
89  {
90  // Reset the interrupt semaphore without waiting
91  xSemaphoreTake(isr_sem, 0);
92  return true;
93  }
94  else
95  {
96  return xSemaphoreTake(isr_sem, timeout_ms / portTICK_PERIOD_MS) == pdTRUE;
97  }
98 }

112 const acc_hal_a121_t *acc_hal_rss_integration_get_implementation(void)
113 {
114  gpio_set_direction(GPIO_SEL0, GPIO_MODE_OUTPUT);
115  gpio_set_direction(GPIO_SEL1, GPIO_MODE_OUTPUT);
116  gpio_set_direction(GPIO_SEL2, GPIO_MODE_OUTPUT);
117  gpio_set_level(GPIO_SEL0, 0);
118  gpio_set_level(GPIO_SEL1, 0);
119  gpio_set_level(GPIO_SEL2, 0);
120  gpio_set_direction(GPIO_ENABLE, GPIO_MODE_OUTPUT);
121 
122  isr_sem = xSemaphoreCreateBinaryStatic(&xSemaphoreBuffer);
123  configASSERT(isr_sem);
124 
125  esp_rom_gpio_pad_select_gpio(GPIO_INTERRUPT);
126  gpio_set_direction(GPIO_INTERRUPT, GPIO_MODE_INPUT);
127  gpio_pulldown_en(GPIO_INTERRUPT);
128  gpio_pullup_dis(GPIO_INTERRUPT);
129  gpio_set_intr_type(GPIO_INTERRUPT, GPIO_INTR_POSEDGE);
130 
131  gpio_install_isr_service(0);
132  gpio_isr_handler_add(GPIO_INTERRUPT, isr_handler, NULL);
133 
134  esp_err_t ret;
135  spi_bus_config_t buscfg = {.miso_io_num = GPIO_MISO,
136  .mosi_io_num = GPIO_MOSI,
137  .sclk_io_num = GPIO_SCLK,
138  .quadwp_io_num = -1,
139  .quadhd_io_num = -1,
140  .max_transfer_sz = SPI_MAX_TRANSFER_SIZE};
141 
142  spi_device_interface_config_t devcfg = {
143  .clock_speed_hz = SPI_TRANSFER_SPEED,
144  .mode = 0,
145  .spics_io_num = GPIO_CS,
146  .queue_size = 7,
147  .pre_cb = NULL,
148  };
149 
150  ret = spi_bus_initialize(SPI2_HOST, &buscfg, SPI_DMA_CH_AUTO);
151  ESP_ERROR_CHECK(ret);
152 
153  ret = spi_bus_add_device(SPI2_HOST, &devcfg, &spi);
154  ESP_ERROR_CHECK(ret);
155 
156  return &hal;
157 }

171 void acc_hal_integration_sensor_enable(acc_sensor_id_t sensor_id)
172 {
173  (void)sensor_id; // Ignore parameter sensor_id
174 
175  gpio_set_level(GPIO_ENABLE, 1);
176 
177  // Wait 2 ms to make sure that the sensor crystal have time to stabilize
178  acc_integration_sleep_us(2000);
179 
180  // Reset the interrupt semaphore
181  xSemaphoreTake(isr_sem, 0);
182 }
